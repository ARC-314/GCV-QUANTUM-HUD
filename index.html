// Remplace la fonction addTx du script précédent par celle-ci pour le LIVE réel async function connectToPiBlockchain() { const log = document.getElementById('log'); const trans = document.getElementById('trans'); // Connexion au flux d'opérations réel du Mainnet Pi const source = new EventSource("https://api.mainnet.minepi.com/operations?cursor=now&order=asc"); source.onmessage = function(event) { const data = JSON.parse(event.data); // On ne filtre que les paiements (transactions de jetons) if (data.type === "payment") { const time = new Date().toLocaleTimeString(); const sender = data.from.substring(0, 4) + "..." + data.from.substring(data.from.length - 4); const amount = parseFloat(data.amount).toFixed(2); // Calcul GCV Réel const gcvTotal = (amount * 314159).toLocaleString('en-US', { minimumFractionDigits: 2 }); const line = document.createElement('div'); line.className = 'tx-line'; line.innerHTML = ` [${time}] FROM: ${sender} | ${amount} PI | GCV VALUE: $${gcvTotal} -> VALIDATED `; log.prepend(line); if (log.childNodes.length > 30) log.removeChild(log.lastChild); // Mise à jour de la transition système let val = parseFloat(trans.innerText); if (val < 100) trans.innerText = (val + 0.0001).toFixed(4) + "%"; } }; source.onerror = function() { console.log("Reconnecting to Pi Mainnet..."); }; } // Lancer la connexion au chargement window.onload = connectToPiBlockchain;
